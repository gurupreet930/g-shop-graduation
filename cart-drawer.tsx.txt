import { useState } from "react";
import { X, Minus, Plus, ShoppingCart, Trash2, Tag } from "lucide-react";
import { useCart } from "@/lib/cart-context";
import { CheckoutModal } from "@/components/checkout-modal";
import { formatCurrency } from "@/lib/utils";
import type { Product } from "@workspace/api-client-react";

export function CartDrawer() {
  const { items, removeItem, updateQty, subtotal, savings, finalTotal, totalItems, isOpen, closeCart } = useCart();
  const [checkoutProduct, setCheckoutProduct] = useState<Product | null>(null);
  const [checkoutOpen, setCheckoutOpen] = useState(false);

  const handleCheckout = () => {
    if (items.length === 0) return;
    // For cart checkout, use first item as the primary product
    setCheckoutProduct(items[0].product);
    setCheckoutOpen(true);
  };

  if (!isOpen) return (
    <>
      <CheckoutModal
        product={checkoutProduct}
        open={checkoutOpen}
        onOpenChange={setCheckoutOpen}
        cartItems={items}
        cartSubtotal={subtotal}
        cartSavings={savings}
        cartFinalTotal={finalTotal}
      />
    </>
  );

  return (
    <>
      {/* Backdrop */}
      <div
        className="fixed inset-0 z-50 bg-black/40"
        onClick={closeCart}
        aria-hidden
      />

      {/* Drawer */}
      <div className="fixed top-0 right-0 z-50 h-full w-full max-w-[420px] bg-white shadow-2xl flex flex-col">
        {/* Header */}
        <div className="flex items-center justify-between px-5 py-4 border-b bg-[#232f3e] text-white">
          <div className="flex items-center gap-2">
            <ShoppingCart className="w-5 h-5" />
            <h2 className="font-bold text-lg">Your Cart ({totalItems})</h2>
          </div>
          <button onClick={closeCart} className="p-1 hover:text-primary transition-colors rounded">
            <X className="w-5 h-5" />
          </button>
        </div>

        {/* Items */}
        <div className="flex-1 overflow-y-auto">
          {items.length === 0 ? (
            <div className="flex flex-col items-center justify-center h-full text-center p-8 text-[#666]">
              <ShoppingCart className="w-16 h-16 mb-4 text-[#ccc]" />
              <p className="font-semibold text-[#333] mb-1">Your cart is empty</p>
              <p className="text-sm">Add items to get started</p>
            </div>
          ) : (
            <ul className="divide-y">
              {items.map((item) => (
                <li key={item.product.id} className="flex gap-3 p-4">
                  <div className="w-20 h-20 bg-[#f8f8f8] rounded flex items-center justify-center flex-shrink-0">
                    {item.product.imageUrl ? (
                      <img src={item.product.imageUrl} alt={item.product.name} className="w-full h-full object-contain p-1" />
                    ) : (
                      <span className="text-2xl font-bold text-[#ccc]">G</span>
                    )}
                  </div>
                  <div className="flex-1 min-w-0">
                    <p className="text-sm font-medium text-[#0F1111] line-clamp-2 leading-snug mb-1">
                      {item.product.name}
                    </p>
                    <p className="text-base font-bold text-[#0F1111] mb-2">
                      {formatCurrency(item.product.salePrice * item.quantity)}
                    </p>
                    <div className="flex items-center gap-3">
                      <div className="flex items-center border border-[#888] rounded-full overflow-hidden">
                        <button
                          onClick={() => updateQty(item.product.id, item.quantity - 1)}
                          className="px-2 py-1 text-[#333] hover:bg-[#f3f3f3] transition-colors"
                        >
                          <Minus className="w-3 h-3" />
                        </button>
                        <span className="px-3 text-sm font-semibold border-x border-[#888]">{item.quantity}</span>
                        <button
                          onClick={() => updateQty(item.product.id, item.quantity + 1)}
                          className="px-2 py-1 text-[#333] hover:bg-[#f3f3f3] transition-colors"
                        >
                          <Plus className="w-3 h-3" />
                        </button>
                      </div>
                      <button
                        onClick={() => removeItem(item.product.id)}
                        className="text-[#c7511f] hover:text-red-600 transition-colors"
                      >
                        <Trash2 className="w-4 h-4" />
                      </button>
                    </div>
                  </div>
                </li>
              ))}
            </ul>
          )}
        </div>

        {/* Summary */}
        {items.length > 0 && (
          <div className="border-t p-5 space-y-3 bg-white">
            {savings > 0 && (
              <div className="flex items-center gap-2 bg-green-50 border border-green-200 rounded px-3 py-2">
                <Tag className="w-4 h-4 text-green-600 flex-shrink-0" />
                <div className="flex-1 text-sm">
                  <span className="text-green-700 font-bold">Instant Savings Applied!</span>
                  <span className="text-green-600 ml-1">You save {formatCurrency(savings)} on this order.</span>
                </div>
              </div>
            )}

            <div className="space-y-1 text-sm">
              <div className="flex justify-between text-[#555]">
                <span>Subtotal ({totalItems} item{totalItems !== 1 ? "s" : ""})</span>
                <span>{formatCurrency(subtotal)}</span>
              </div>
              {savings > 0 && (
                <div className="flex justify-between text-green-600 font-medium">
                  <span>Instant Savings</span>
                  <span>-{formatCurrency(savings)}</span>
                </div>
              )}
              <div className="flex justify-between font-bold text-base text-[#0F1111] pt-2 border-t">
                <span>Order Total</span>
                <span>{formatCurrency(finalTotal)}</span>
              </div>
            </div>

            {subtotal <= 100 && (
              <p className="text-xs text-[#555] bg-[#fff8e6] border border-[#f0a030] rounded px-3 py-2">
                Add {formatCurrency(100 - subtotal + 0.01)} more to unlock{" "}
                <strong>$30 Instant Savings</strong>. Add{" "}
                {formatCurrency(200 - subtotal + 0.01)} more for <strong>$50 savings</strong>.
              </p>
            )}
            {subtotal > 100 && subtotal <= 200 && (
              <p className="text-xs text-[#555] bg-[#fff8e6] border border-[#f0a030] rounded px-3 py-2">
                Add {formatCurrency(200 - subtotal + 0.01)} more to upgrade to <strong>$50 Instant Savings</strong>!
              </p>
            )}

            <button
              onClick={handleCheckout}
              className="w-full h-11 bg-primary hover:bg-[#f0a130] active:bg-[#e08c1a] transition-colors text-black font-bold rounded-full border border-[#d5a429] shadow-sm text-sm"
            >
              Proceed to Checkout
            </button>
            <button
              onClick={closeCart}
              className="w-full h-9 bg-white hover:bg-[#f3f3f3] transition-colors text-[#333] font-medium rounded-full border border-[#ccc] text-sm"
            >
              Continue Shopping
            </button>
          </div>
        )}
      </div>

      <CheckoutModal
        product={checkoutProduct}
        open={checkoutOpen}
        onOpenChange={setCheckoutOpen}
        cartItems={items}
        cartSubtotal={subtotal}
        cartSavings={savings}
        cartFinalTotal={finalTotal}
      />
    </>
  );
}
