import { Link } from "wouter";
import { ShoppingCart, Zap } from "lucide-react";
import type { Product } from "@workspace/api-client-react";
import { StarRating, productRating } from "@/components/star-rating";
import { useCart } from "@/lib/cart-context";
import { formatCurrency } from "@/lib/utils";

interface ProductCardProps {
  product: Product;
  onBuyNow?: (product: Product) => void;
}

export function ProductCard({ product, onBuyNow }: ProductCardProps) {
  const { addItem } = useCart();
  const { rating, reviews } = productRating(product.id);

  return (
    <div className="bg-white rounded-sm shadow-sm hover:shadow-md transition-shadow duration-200 flex flex-col h-full group">
      {/* Image */}
      <Link href={`/product/${product.id}`} className="block relative overflow-hidden bg-white rounded-t-sm pt-4 px-4">
        {product.badge && (
          <span className="absolute top-2 left-2 z-10 bg-primary text-black text-[10px] font-bold px-2 py-0.5 rounded-sm">
            {product.badge}
          </span>
        )}
        <div className="aspect-square flex items-center justify-center">
          {product.imageUrl ? (
            <img
              src={product.imageUrl}
              alt={product.name}
              className="w-full h-full object-contain transition-transform duration-300 group-hover:scale-105"
            />
          ) : (
            <div className="w-full h-full bg-[#f3f3f3] flex items-center justify-center text-[#aaa] text-5xl font-bold">
              G
            </div>
          )}
        </div>
      </Link>

      {/* Details */}
      <div className="p-3 flex flex-col flex-1">
        <Link href={`/product/${product.id}`} className="block mb-1">
          <h3 className="text-sm font-medium text-[#0F1111] leading-snug line-clamp-2 hover:text-[#c7511f] transition-colors">
            {product.name}
          </h3>
        </Link>

        <StarRating rating={rating} reviewCount={reviews} size="sm" />

        {/* Price — only final selling price shown */}
        <div className="mt-2 mb-3">
          <div className="flex items-baseline gap-1">
            <span className="text-xs text-[#0F1111] align-top leading-none mt-1">$</span>
            <span className="text-2xl font-medium text-[#0F1111] leading-none">
              {Math.floor(product.salePrice)}
            </span>
            <span className="text-xs text-[#0F1111] leading-none mt-1">
              {(product.salePrice % 1).toFixed(2).slice(1)}
            </span>
          </div>
          {product.stock <= 5 && product.stock > 0 && (
            <p className="text-xs text-[#c7511f] font-medium mt-1">Only {product.stock} left in stock</p>
          )}
        </div>

        {/* CTA buttons */}
        <div className="mt-auto space-y-2">
          <button
            onClick={() => addItem(product)}
            className="w-full h-9 bg-primary hover:bg-[#f0a130] active:bg-[#e08c1a] transition-colors text-black text-sm font-medium rounded-full border border-[#d5a429] shadow-sm flex items-center justify-center gap-2"
          >
            <ShoppingCart className="w-4 h-4" />
            Add to Cart
          </button>
          <button
            onClick={() => onBuyNow?.(product)}
            className="w-full h-9 bg-[#FFD814] hover:bg-[#F7CA00] active:bg-[#DDB100] transition-colors text-black text-sm font-medium rounded-full border border-[#FCD200] shadow-sm flex items-center justify-center gap-2"
          >
            <Zap className="w-4 h-4" />
            Buy Now
          </button>
        </div>
      </div>
    </div>
  );
}
