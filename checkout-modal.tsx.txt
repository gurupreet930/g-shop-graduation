import { useState } from "react";
import { useLocation } from "wouter";
import { z } from "zod";
import { useForm } from "react-hook-form";
import { zodResolver } from "@hookform/resolvers/zod";
import { useCreateOrder, getListOrdersQueryKey } from "@workspace/api-client-react";
import type { Product } from "@workspace/api-client-react";
import { useQueryClient } from "@tanstack/react-query";
import { Dialog, DialogContent, DialogHeader, DialogTitle, DialogDescription } from "@/components/ui/dialog";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Form, FormControl, FormField, FormItem, FormLabel, FormMessage } from "@/components/ui/form";
import { useToast } from "@/hooks/use-toast";
import { formatCurrency } from "@/lib/utils";
import { Check, Tag, ShieldCheck, Truck } from "lucide-react";
import type { CartItem } from "@/lib/cart-context";
import { useCart } from "@/lib/cart-context";

const checkoutSchema = z.object({
  customerName: z.string().min(2, "Name is required"),
  customerEmail: z.string().email("Invalid email address"),
  customerAddress: z.string().min(5, "Address is required"),
  quantity: z.number().min(1).max(10),
});

type CheckoutFormValues = z.infer<typeof checkoutSchema>;

interface CheckoutModalProps {
  product: Product | null;
  open: boolean;
  onOpenChange: (open: boolean) => void;
  cartItems?: CartItem[];
  cartSubtotal?: number;
  cartSavings?: number;
  cartFinalTotal?: number;
}

export function CheckoutModal({
  product,
  open,
  onOpenChange,
  cartItems,
  cartSubtotal,
  cartSavings = 0,
  cartFinalTotal,
}: CheckoutModalProps) {
  const { toast } = useToast();
  const [, setLocation] = useLocation();
  const queryClient = useQueryClient();
  const { clearCart } = useCart();
  const [isSuccess, setIsSuccess] = useState(false);
  const [orderId, setOrderId] = useState<number | null>(null);

  const isCartCheckout = !!cartItems && cartItems.length > 0;

  const form = useForm<CheckoutFormValues>({
    resolver: zodResolver(checkoutSchema),
    defaultValues: {
      customerName: "",
      customerEmail: "",
      customerAddress: "",
      quantity: 1,
    },
  });

  const quantity = form.watch("quantity");
  const singleTotal = product ? product.salePrice * quantity : 0;

  // For single-product checkout compute savings inline
  const singleSavings = !isCartCheckout
    ? (singleTotal > 200 ? 50 : singleTotal > 100 ? 30 : 0)
    : 0;

  const displaySubtotal = isCartCheckout ? (cartSubtotal ?? 0) : singleTotal;
  const displaySavings = isCartCheckout ? cartSavings : singleSavings;
  const displayTotal = isCartCheckout ? (cartFinalTotal ?? 0) : singleTotal - singleSavings;

  const createOrder = useCreateOrder({
    mutation: {
      onSuccess: (data) => {
        queryClient.invalidateQueries({ queryKey: getListOrdersQueryKey() });
        setOrderId(data.id);
        setIsSuccess(true);
        if (isCartCheckout) clearCart();
      },
      onError: () => {
        toast({
          title: "Order Failed",
          description: "There was an issue processing your order. Please try again.",
          variant: "destructive",
        });
      },
    },
  });

  const onSubmit = (values: CheckoutFormValues) => {
    if (!product) return;
    createOrder.mutate({
      data: {
        productId: product.id,
        customerName: values.customerName,
        customerEmail: values.customerEmail,
        customerAddress: values.customerAddress,
        quantity: isCartCheckout ? (cartItems?.reduce((s, i) => s + i.quantity, 0) ?? 1) : values.quantity,
      },
    });
  };

  const handleClose = () => {
    onOpenChange(false);
    setTimeout(() => {
      if (isSuccess) {
        setIsSuccess(false);
        form.reset();
        setLocation("/orders");
      }
    }, 300);
  };

  if (!product) return null;

  return (
    <Dialog open={open} onOpenChange={handleClose}>
      <DialogContent className="sm:max-w-[520px] border-[#ddd] bg-white p-0 overflow-hidden rounded-sm">
        {isSuccess ? (
          <div className="p-10 text-center flex flex-col items-center">
            <div className="w-16 h-16 rounded-full bg-green-100 flex items-center justify-center mb-4">
              <Check className="w-8 h-8 text-green-600" />
            </div>
            <DialogTitle className="text-2xl font-bold text-[#0F1111] mb-2">Order Placed!</DialogTitle>
            <DialogDescription className="text-base text-[#555] mb-2">
              Order #{orderId} has been confirmed.
            </DialogDescription>
            <p className="text-sm text-[#555] mb-6">
              Estimated delivery: <strong className="text-[#0F1111]">2–4 business days</strong>
            </p>

            {displaySavings > 0 && (
              <div className="w-full mb-6 bg-green-50 border border-green-200 rounded p-4 text-left">
                <div className="flex items-center gap-2 mb-1">
                  <Tag className="w-4 h-4 text-green-600" />
                  <span className="text-sm font-bold text-green-700">You Saved {formatCurrency(displaySavings)}!</span>
                </div>
                <p className="text-xs text-green-600">
                  Instant Savings were automatically applied to your order total.
                </p>
              </div>
            )}

            <Button
              onClick={handleClose}
              className="w-full h-11 bg-primary hover:bg-[#f0a130] text-black font-bold rounded-full border border-[#d5a429] text-sm"
            >
              Track Your Order
            </Button>
          </div>
        ) : (
          <>
            <div className="px-6 pt-5 pb-4 border-b border-[#ddd] bg-[#f8f8f8]">
              <DialogHeader>
                <DialogTitle className="text-xl font-bold text-[#0F1111]">Checkout</DialogTitle>
                <DialogDescription className="text-sm text-[#555]">
                  {isCartCheckout
                    ? `${cartItems?.length} item${cartItems?.length !== 1 ? "s" : ""} in your cart`
                    : `Purchasing: ${product.name}`}
                </DialogDescription>
              </DialogHeader>

              {/* Cart items preview or single product */}
              {isCartCheckout ? (
                <div className="mt-4 space-y-2 max-h-32 overflow-y-auto">
                  {cartItems?.map((item) => (
                    <div key={item.product.id} className="flex items-center gap-3 bg-white border border-[#ddd] rounded p-2">
                      <div className="w-10 h-10 bg-[#f3f3f3] flex-shrink-0 flex items-center justify-center rounded">
                        {item.product.imageUrl
                          ? <img src={item.product.imageUrl} alt={item.product.name} className="w-full h-full object-contain p-0.5" />
                          : <span className="text-[#aaa] font-bold">G</span>}
                      </div>
                      <div className="flex-1 min-w-0">
                        <p className="text-xs font-medium text-[#0F1111] line-clamp-1">{item.product.name}</p>
                        <p className="text-xs text-[#555]">Qty: {item.quantity}</p>
                      </div>
                      <p className="text-xs font-bold text-[#0F1111] flex-shrink-0">
                        {formatCurrency(item.product.salePrice * item.quantity)}
                      </p>
                    </div>
                  ))}
                </div>
              ) : (
                <div className="mt-4 flex items-center gap-3 bg-white border border-[#ddd] rounded p-3">
                  <div className="w-14 h-14 bg-[#f3f3f3] flex-shrink-0 flex items-center justify-center rounded">
                    {product.imageUrl
                      ? <img src={product.imageUrl} alt={product.name} className="w-full h-full object-contain p-1" />
                      : <span className="text-[#aaa] font-bold text-xl">G</span>}
                  </div>
                  <div className="flex-1">
                    <p className="text-sm font-semibold text-[#0F1111] line-clamp-1">{product.name}</p>
                    <p className="text-primary font-bold">{formatCurrency(product.salePrice)}</p>
                  </div>
                </div>
              )}
            </div>

            <div className="px-6 py-5">
              <Form {...form}>
                <form onSubmit={form.handleSubmit(onSubmit)} className="space-y-4">
                  <div className="grid grid-cols-1 gap-4">
                    <FormField
                      control={form.control}
                      name="customerName"
                      render={({ field }) => (
                        <FormItem>
                          <FormLabel className="text-xs font-bold text-[#333] uppercase tracking-wide">Full Name</FormLabel>
                          <FormControl>
                            <Input placeholder="John Doe" className="border-[#888] focus-visible:ring-primary bg-white text-[#0F1111]" {...field} />
                          </FormControl>
                          <FormMessage />
                        </FormItem>
                      )}
                    />
                    <FormField
                      control={form.control}
                      name="customerEmail"
                      render={({ field }) => (
                        <FormItem>
                          <FormLabel className="text-xs font-bold text-[#333] uppercase tracking-wide">Email Address</FormLabel>
                          <FormControl>
                            <Input type="email" placeholder="john@example.com" className="border-[#888] focus-visible:ring-primary bg-white text-[#0F1111]" {...field} />
                          </FormControl>
                          <FormMessage />
                        </FormItem>
                      )}
                    />
                    <FormField
                      control={form.control}
                      name="customerAddress"
                      render={({ field }) => (
                        <FormItem>
                          <FormLabel className="text-xs font-bold text-[#333] uppercase tracking-wide">Shipping Address</FormLabel>
                          <FormControl>
                            <Input placeholder="123 Main St, New York, NY 10001" className="border-[#888] focus-visible:ring-primary bg-white text-[#0F1111]" {...field} />
                          </FormControl>
                          <FormMessage />
                        </FormItem>
                      )}
                    />

                    {!isCartCheckout && (
                      <FormField
                        control={form.control}
                        name="quantity"
                        render={({ field }) => (
                          <FormItem>
                            <FormLabel className="text-xs font-bold text-[#333] uppercase tracking-wide">Quantity</FormLabel>
                            <FormControl>
                              <div className="flex items-center gap-0">
                                <button type="button" onClick={() => field.onChange(Math.max(1, field.value - 1))}
                                  className="h-9 w-9 border border-[#888] rounded-l bg-[#f3f3f3] hover:bg-[#e8e8e8] text-[#333] font-bold transition-colors">-</button>
                                <Input type="number" min={1} max={10}
                                  className="h-9 rounded-none text-center border-y border-x-0 border-[#888] focus-visible:ring-primary bg-white text-[#0F1111] w-16"
                                  {...field} onChange={(e) => field.onChange(parseInt(e.target.value) || 1)} />
                                <button type="button" onClick={() => field.onChange(Math.min(10, field.value + 1))}
                                  className="h-9 w-9 border border-[#888] rounded-r bg-[#f3f3f3] hover:bg-[#e8e8e8] text-[#333] font-bold transition-colors">+</button>
                              </div>
                            </FormControl>
                            <FormMessage />
                          </FormItem>
                        )}
                      />
                    )}
                  </div>

                  {/* Order Summary */}
                  <div className="bg-[#f8f8f8] border border-[#ddd] rounded p-4 space-y-2 text-sm">
                    <div className="flex justify-between text-[#555]">
                      <span>Items:</span>
                      <span>{formatCurrency(displaySubtotal)}</span>
                    </div>
                    <div className="flex justify-between text-[#555]">
                      <span>Shipping:</span>
                      <span className="text-green-600 font-medium">FREE</span>
                    </div>
                    {displaySavings > 0 && (
                      <div className="flex justify-between text-green-600 font-bold">
                        <span className="flex items-center gap-1">
                          <Tag className="w-3 h-3" /> Instant Savings:
                        </span>
                        <span>-{formatCurrency(displaySavings)}</span>
                      </div>
                    )}
                    <div className="border-t border-[#ddd] pt-2 flex justify-between font-bold text-[#0F1111] text-base">
                      <span>Order Total:</span>
                      <span>{formatCurrency(displayTotal)}</span>
                    </div>
                    {displaySubtotal <= 100 && displaySubtotal > 0 && (
                      <p className="text-xs text-[#c7511f] pt-1">
                        Add {formatCurrency(100 - displaySubtotal + 0.01)} more for <strong>$30 Instant Savings</strong>
                      </p>
                    )}
                    {displaySubtotal > 100 && displaySubtotal <= 200 && (
                      <p className="text-xs text-[#c7511f] pt-1">
                        Add {formatCurrency(200 - displaySubtotal + 0.01)} more to upgrade to <strong>$50 Instant Savings</strong>
                      </p>
                    )}
                  </div>

                  <div className="flex items-center gap-3 text-xs text-[#555]">
                    <div className="flex items-center gap-1"><ShieldCheck className="w-3.5 h-3.5 text-green-600" /> Secure checkout</div>
                    <div className="flex items-center gap-1"><Truck className="w-3.5 h-3.5 text-primary" /> Free shipping</div>
                  </div>

                  <Button
                    type="submit"
                    className="w-full h-11 text-sm font-bold bg-primary hover:bg-[#f0a130] active:bg-[#e08c1a] text-black rounded-full border border-[#d5a429] shadow-sm"
                    disabled={createOrder.isPending}
                  >
                    {createOrder.isPending ? "Placing Order..." : "Place Your Order"}
                  </Button>
                </form>
              </Form>
            </div>
          </>
        )}
      </DialogContent>
    </Dialog>
  );
}
